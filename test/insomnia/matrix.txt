# Insomnia's guard matrix, ported from Insomnia's test.sh line by line, plus the 2.0 checks:
# one pair of assertions per working agent's pid, the end on that pid's exit, the end on low battery,
# holds a script asks for, the standalone app, failed writes and the sudoers rule.
# test/insomnia/tool.m runs it. Insomnia's debug prefs become the fake power source, the clock and fixture workers.

section self-heal
# MinMacs set the override and then crashed: the next launch clears it, as Insomnia's launch did.
pref insomnia.lid.ours true
fake lid 1
fake rule on
step
expect lid paused a stale override MinMacs set is cleared at launch
expect undo 1 the write has an undo record

section toggle
pref insomnia.agents false
step
expect asserts 0 off releases assertions
set manual on
expect asserts 2 on holds two assertions
expect held 1 on holds one pair, for MinMacs itself
expect types PreventSystemSleep,PreventUserIdleSystemSleep the pair is both of Insomnia's assertion types
expect state awake,_Keep_Mac_Awake_is_on the status item says so
set manual off
expect asserts 0 off releases them again

section awake while agents work
set agents on
worker a start
step
expect asserts 2 mode on, 1 working, manual off: holds
expect holdfor:a yes the pair is tied to the working agent's pid
expect types PreventSystemSleep,PreventUserIdleSystemSleep an agent's pair is both types too
worker a idle
clock +5
step
expect asserts 2 0 working, idle 5 min, quiet 10: still holds
clock +6
step
expect asserts 0 0 working, idle 11 min, quiet 10: releases
set quiet 20
expect asserts 2 quiet period 20: idle 11 min holds again
set quiet 10
expect asserts 0 back to the default 10 min: releases
worker a working
step
expect asserts 2 agents start working again: holds
worker a idle
clock +11
step
set manual on
expect asserts 2 manual on wins over agents idle
expect holdfor:a no manual on holds for MinMacs, not for the agent
set manual off
expect asserts 0 manual off, agents idle: releases
worker a working
set manual on
set manual off
expect asserts 2 manual off while agents work: agent mode holds
set agents off
expect asserts 0 mode off, agent still working: releases
worker a idle
set agents on
expect asserts 0 mode off clears the working clock
set agents off
set manual on
expect asserts 2 mode off leaves manual on alone
set manual off
expect asserts 0 mode off, manual off: as before

section one pair per agent's pid
set agents on
worker a working
worker b start
step
expect held 2 two working agents hold two pairs
expect asserts 4 four assertions in IOKit
worker b exit
wait held 1
expect held 1 a pair ends when its agent's pid exits, before the next pass
expect asserts 2 and IOKit drops its two assertions
step
expect holdfor:a yes the other agent keeps its pair

section low battery
power onAC=0 battery=50
step
expect asserts 2 on battery at 50%: the agent holds
power battery=20
step
expect asserts 0 battery 20%, the threshold: agent holds end
expect lowbattery yes the state says why
set manual on
expect asserts 2 manual on still holds at low battery
set manual off
power onAC=1 battery=21
step
expect asserts 2 charger back: the agent holds again

section holds a script asks for
worker a idle
clock +11
step
expect asserts 0 nothing works
proc h start
hold h a build runs
step
expect holdfor:h yes a hold a script asks for keeps the Mac awake
proc h exit
wait held 0
expect held 0 it ends when that process exits
unhold h
proc g start
cli insomnia hold $g render runs
expect cli-status 0 the hold verb runs in the app
waitgraph
step
expect holdfor:g yes the graph carries it to Insomnia
proc g exit
wait held 0
expect held 0 and it ends on that exit too
step
